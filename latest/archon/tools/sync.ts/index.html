archon/tools/sync.ts

archon tools sync (spec 008, Articles V and XII).

upstream-binary: latest GitHub release -> verify the asset's published SHA-256 digest -> install under /opt/tools/<id>/<version> -> smoke test -> atomically switch /usr/local/bin/<bin> -> keep the previous version. apt: installed by apt; reported against the candidate in Ubuntu's archive. Nothing is ever built from source on a host: binaries only. self-updating: install for persona holders, then record versions only.

export interface Release {
  /** The release's git tag, as published (for example `"v2.1.0"`). */
  tag: string;
  /** The version: the tag without the tool's tag prefix. */
  version: string;
  /** The release's downloadable files, each with its published digest when GitHub has one. */
  assets: Array<{
    /** The file name. */
    name: string;
    /** Where to download it. */
    url: string;
    /** GitHub's digest for it, `sha256:<hex>`, when published. */
    digest?: string;
  }>;
}

  /** What the run did or would do (`"current"`, `"upgraded a -> b"`, `"FAILED: ..."`, and so on). */
  action: string;

}
