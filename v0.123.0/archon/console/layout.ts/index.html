archon/console/layout.ts

Specs 065 and 066: both consoles' frame. A top bar (Archon on the left, the fleet with Stolos in the middle, the signed-in person on the right), a menu of pages, and the page. Datastar makes parts of a page live; Mermaid draws the diagrams (from the page's own data), loaded only on pages that have one, and drawn again when Datastar swaps one in.

interfaceFrame

export interface Frame {
  /** "archon" (one machine) or "stolos" (the fleet). */
  product: "archon" | "stolos";
  /** The page's title (also the tab's). */
  title: string;
  /** Under the product's name: the machine and its OS, or the PC. */
  subtitle: string;
  /** The fleet and its repository, in the middle. */
  fleet: { id: string; repo: string };
  /** The person signed in: name, username, and role in words. */
  user: { name: string; user: string; role: string };
  /** The menu. */
  nav: NavItem[];
  /** The current page's href. */
  current: string;
  /** Spec 082: the provider's client this page is for, and where the list of clients is. */
  scope?: { label: string; href: string };
}
