device

Spec 069: devices: printers, routers, switches, access points, NAS boxes, cameras, UPSes. They run no script, so Stolos reads them over SNMPv3, read-only and encrypted (authPriv), with a user made on the device for it. Each is declared in fleets/<fleet>/devices/<name>/device.ts; what it said is in observed.ts (GENERATED; git log -p is its history). Its passphrases are in the vault (devices/<name>/snmp.json), never in Git.

Source: src/archon/device/device.ts

const DEVICE_KINDS: readonly ["router", "switch", "access-point", "firewall", "printer", "nas", "camera", "ups", "other"]

type DeviceDef = {
  /** Its name in the fleet (a DNS label). */
  name: string;
  /** What kind of device. */
  kind: (typeof DEVICE_KINDS)[number];
  /** Where Stolos reaches it (an address or a name). */
  address: string;
  /** Its SNMP port (161 unless set). */
  port?: number;
  /** What it is, for people ("Front desk printer"). */
  title?: string;
  /** Who is responsible for it. */
  owner?: string;
  /** Where it is (a room, a rack). */
  location?: string;
  /** The SNMPv3 user Stolos reads it as, and the authentication protocol (privacy is always AES). */
  snmp: { user: string; auth: "sha" | "sha256" };
  /** When it was added (ISO time). */
  added: string;
};
